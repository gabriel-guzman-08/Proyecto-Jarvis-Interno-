import pychrome


browser = pychrome.Browser(
    url="http://127.0.0.1:9222"
)


def abrir_youtube():

    tabs = browser.list_tab()

    if tabs:

        tab = tabs[0]

    else:

        tab = browser.new_tab()

    tab.start()

    tab.Page.enable()

    tab.Page.navigate(
        url="https://youtube.com"
    )

    tab.wait(3)

    tab.stop()
